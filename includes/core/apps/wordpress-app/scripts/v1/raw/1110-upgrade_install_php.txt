#!/bin/bash
# This script will install PHP ${phpver}

echo $(date): "Refreshing repositories..."
apt-get update > /dev/null 2>&1
if [ $? -ne 0 ]
then
	echo "Failed!  Quitting process"
	exit 1
fi

## Get our common functions
if [[ ! -f 9999-common-functions.sh ]]
then
	echo "The file 9999-common-functions.sh is missing"
	exit 1
fi
source 9999-common-functions.sh

if [[ -z $phpver ]]; then
	echo
	echo "Specify PHP version You Would Like Installed"
	read -p "PHP Version: " phpver
fi

phpver_compact="${phpver//./}"

echo $(date): "Installing PHP ${phpver}...."
export DEBIAN_FRONTEND=noninteractive
# Add Vhost config based on webserver_type
if [ "$g_webserver_type" = "nginx" ]; then
	# Keep this extension set the same as the one 01-prepare_server.txt installs
	# for the baseline versions, so a version added later is not missing
	# anything the ones installed at provisioning time have.
	apt-get install -y -o Dpkg::Options::="--force-confdef" -o Dpkg::Options::="--force-confold"  php${phpver} php${phpver}-fpm php${phpver}-mbstring php${phpver}-curl php${phpver}-mysql php${phpver}-xml php${phpver}-zip php${phpver}-gd php${phpver}-imap php${phpver}-soap php${phpver}-bcmath php${phpver}-imagick php${phpver}-intl -y
	if [ $? -ne 0 ]
	then
		echo "Failed!  Quitting process"
		exit 1
	fi

	# Redis object caching, matching what 01-prepare_server.txt installs for the
	# baseline versions.  Not fatal - the server still works without it.
	if [ -f /etc/redis/redis.conf ]
	then
		echo $(date): "Adding the Redis extension..."
		apt-get install -y php${phpver}-redis > /dev/null 2>&1 || echo $(date): "Could not install the Redis extension for PHP ${phpver}."
	fi

	# Update php.ini file to increase filesize uploads allowed in WordPress
	echo $(date): "Adding required entries in php.ini to allow for larger file uploads in WordPress..."
	sed -i "s/upload_max_filesize = 2M/upload_max_filesize = 25M/g" /etc/php/${phpver}/fpm/php.ini
	sed -i "s/post_max_size = 8M/post_max_size = 25M/g" /etc/php/${phpver}/fpm/php.ini

	# Update php.ini to enable and configure opcache
	echo $(date): "Updating OPCACHE parameters in php.ini..."
	sed -i "s/;opcache.enable=1/opcache.enable=1/g" /etc/php/${phpver}/fpm/php.ini
	sed -i "s/;opcache.enable_cli=0/opcache.enable_cli=1/g" /etc/php/${phpver}/fpm/php.ini
	sed -i "s/;opcache.interned_strings_buffer=8/opcache.interned_strings_buffer=8/g" /etc/php/${phpver}/fpm/php.ini
	sed -i "s/;opcache.max_accelerated_files=10000/opcache.max_accelerated_files=100000/g" /etc/php/${phpver}/fpm/php.ini
	sed -i "s/;opcache.memory_consumption=128/opcache.memory_consumption=128/g" /etc/php/${phpver}/fpm/php.ini
	sed -i "s/;opcache.save_comments=1/opcache.save_comments=1/g" /etc/php/${phpver}/fpm/php.ini
	sed -i "s/;opcache.revalidate_freq=2/opcache.revalidate_freq=1/g" /etc/php/${phpver}/fpm/php.ini
	sed -i "s/;opcache.use_cwd=1/opcache.use_cwd=1/g" /etc/php/${phpver}/fpm/php.ini
	sed -i "s/;opcache.validate_root=0/opcache.validate_root=1/g" /etc/php/${phpver}/fpm/php.ini
	sed -i "s/;opcache.validate_permission=0/opcache.validate_permission=1/g" /etc/php/${phpver}/fpm/php.ini
	sed -i "s/;opcache.file_update_protection=2/opcache.file_update_protection=60/g" /etc/php/${phpver}/fpm/php.ini

	# PHP ${phpver} needs a set of default functions added to its global php.ini file because its set to blank for some reason.
	# We're going to use the same list as 7.4.
	sed -i "s/^disable_functions.*/disable_functions = pcntl_alarm,pcntl_fork,pcntl_waitpid,pcntl_wait,pcntl_wifexited,pcntl_wifstopped,pcntl_wifsignaled,pcntl_wifcontinued,pcntl_wexitstatus,pcntl_wtermsig,pcntl_wstopsig,pcntl_signal,pcntl_signal_get_handler,pcntl_signal_dispatch,pcntl_get_last_error,pcntl_strerror,pcntl_sigprocmask,pcntl_sigwaitinfo,pcntl_sigtimedwait,pcntl_exec,pcntl_getpriority,pcntl_setpriority,pcntl_async_signals,pcntl_unshare,/g" /etc/php/${phpver}/fpm/php.ini

	# Restarting php
	echo $(date): "Restarting PHP processes..."
	if [ ! -f /etc/wpcd/php-versions-disabled/php${phpver} ] ; then
		systemctl restart php${phpver}-fpm > /dev/null 2>&1
	fi

# OLS Wrapper
elif [ "$g_webserver_type" = "ols" ]  || [ "$g_webserver_type" = "ols-enterprise" ]; then
	if [ "$phpver" = "8.1" ]; then
		apt-get install -y lsphp${phpver_compact} lsphp${phpver_compact}-common lsphp${phpver_compact}-dev lsphp${phpver_compact}-pear lsphp${phpver_compact}-mysql lsphp${phpver_compact}-imap lsphp${phpver_compact}-curl lsphp${phpver_compact}-memcached lsphp${phpver_compact}-redis lsphp*
	else
		apt-get install -y lsphp${phpver_compact} lsphp${phpver_compact}-common lsphp${phpver_compact}-dev lsphp${phpver_compact}-pear lsphp${phpver_compact}-mysql lsphp${phpver_compact}-imap lsphp${phpver_compact}-curl
		if [ $? -ne 0 ]
		then
			echo "Failed!  Quitting process"
			exit 1
		fi

		apt-get install -y lsphp${phpver_compact}-imagick lsphp${phpver_compact}-intl lsphp*
		if [ $? -ne 0 ]
		then
			echo "Failed!  Quitting process"
			exit 1
		fi

		sed -i -e "s|^;session.save_path.*|session.save_path = '/var/lib/lsphp/session/lsphp${phpver_compact}'|g" -e "s|^session.save_path.*|session.save_path = '/var/lib/lsphp/session/lsphp${phpver_compact}'|g" /usr/local/lsws/lsphp${phpver_compact}/etc/php/${phpver}/litespeed/php.ini ;

		sed -i 's/^memory_limit.*/memory_limit = 1024M/g' /usr/local/lsws/lsphp${phpver_compact}/etc/php/${phpver}/litespeed/php.ini
		sed -i 's/^post_max_size.*/post_max_size = 25M/g' /usr/local/lsws/lsphp${phpver_compact}/etc/php/${phpver}/litespeed/php.ini
		sed -i 's/^upload_max_filesize.*/upload_max_filesize = 25M/g' /usr/local/lsws/lsphp${phpver_compact}/etc/php/${phpver}/litespeed/php.ini
		sed -i 's/^max_execution_time.*/max_execution_time = 7200/g' /usr/local/lsws/lsphp${phpver_compact}/etc/php/${phpver}/litespeed/php.ini

		sed -i "s/;opcache.enable=1/opcache.enable=1/g" /usr/local/lsws/lsphp${phpver_compact}/etc/php/${phpver}/litespeed/php.ini
		sed -i "s/;opcache.enable_cli=0/opcache.enable_cli=1/g" /usr/local/lsws/lsphp${phpver_compact}/etc/php/${phpver}/litespeed/php.ini
		sed -i "s/;opcache.interned_strings_buffer=8/opcache.interned_strings_buffer=8/g" /usr/local/lsws/lsphp${phpver_compact}/etc/php/${phpver}/litespeed/php.ini
		sed -i "s/;opcache.max_accelerated_files=10000/opcache.max_accelerated_files=100000/g" /usr/local/lsws/lsphp${phpver_compact}/etc/php/${phpver}/litespeed/php.ini
		sed -i "s/;opcache.memory_consumption=128/opcache.memory_consumption=128/g" /usr/local/lsws/lsphp${phpver_compact}/etc/php/${phpver}/litespeed/php.ini
		sed -i "s/;opcache.save_comments=1/opcache.save_comments=1/g" /usr/local/lsws/lsphp${phpver_compact}/etc/php/${phpver}/litespeed/php.ini
		sed -i "s/;opcache.revalidate_freq=2/opcache.revalidate_freq=1/g" /usr/local/lsws/lsphp${phpver_compact}/etc/php/${phpver}/litespeed/php.ini
		sed -i "s/;opcache.use_cwd=1/opcache.use_cwd=1/g" /usr/local/lsws/lsphp${phpver_compact}/etc/php/${phpver}/litespeed/php.ini
		sed -i "s/;opcache.validate_root=0/opcache.validate_root=1/g" /usr/local/lsws/lsphp${phpver_compact}/etc/php/${phpver}/litespeed/php.ini
		sed -i "s/;opcache.validate_permission=0/opcache.validate_permission=1/g" /usr/local/lsws/lsphp${phpver_compact}/etc/php/${phpver}/litespeed/php.ini
		sed -i "s/;opcache.file_update_protection=2/opcache.file_update_protection=60/g" /usr/local/lsws/lsphp${phpver_compact}/etc/php/${phpver}/litespeed/php.ini

		sed -i "s/^disable_functions.*/disable_functions = pcntl_alarm,pcntl_fork,pcntl_waitpid,pcntl_wait,pcntl_wifexited,pcntl_wifstopped,pcntl_wifsignaled,pcntl_wifcontinued,pcntl_wexitstatus,pcntl_wtermsig,pcntl_wstopsig,pcntl_signal,pcntl_signal_get_handler,pcntl_signal_dispatch,pcntl_get_last_error,pcntl_strerror,pcntl_sigprocmask,pcntl_sigwaitinfo,pcntl_sigtimedwait,pcntl_exec,pcntl_getpriority,pcntl_setpriority,pcntl_async_signals,pcntl_unshare,/g" /usr/local/lsws/lsphp${phpver_compact}/etc/php/${phpver}/litespeed/php.ini

		/usr/local/lsws/lsphp${phpver_compact}/bin/pecl channel-update pecl.php.net

		sed -i -e 's|^;date.timezone.*|date.timezone = "UTC"|g' /usr/local/lsws/lsphp${phpver_compact}/etc/php/${phpver}/litespeed/php.ini;
		php_utc_result=$(/usr/local/lsws/lsphp${phpver_compact}/bin/php -i 2>&1 | grep -Ei 'date.timezone')
		if ! echo "$php_utc_result"| grep UTC >/dev/null 2>&1
		then
			echo "Configuring Timezone for UTC for LSPHP${phpver_compact} failed"
			echo "Found: $php_utc_result"
		fi
	fi

	if [ $? -ne 0 ]
	then
		echo "Failed!  Quitting process"
		exit 1
	fi

fi

gf_restart_webserver "true"
echo "PHP ${phpver} has been installed."
